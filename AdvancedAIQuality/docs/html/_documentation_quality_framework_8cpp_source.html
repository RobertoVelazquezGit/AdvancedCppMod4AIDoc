35 AIDocumentationAnalyzer::analyzeDocumentation(const std::string& documentPath) {
36 AnalysisReport report{};
37 if (documentPath.empty()) {
38 report.identifiedIssues.push_back("Empty document label.");
39 return report;
40 }
41 const std::string fixture = "Overview\nUsage\nMock documentation example.";
42 report.missingElements = extractMissingElements(fixture);
43 for (const auto& section : currentConfig.requiredSections)
44 report.sectionScores[section] = fixture.find(section) != std::string::npos ? 1.0 : 0.0;
45 const double completeness = currentConfig.requiredSections.empty() ? 1.0 :
46 1.0 - static_cast<double>(report.missingElements.size()) / currentConfig.requiredSections.size();
47 report.overallMetrics = { calculateTechnicalAccuracy(fixture, "mock source"),
48 completeness, 0.8, 0.85, 0.75, currentConfig.analyzeAudienceAlignment ? 0.8 : 0.0 };
49 for (const auto& missing : report.missingElements) {
50 report.identifiedIssues.push_back("Missing section: " + missing);
51 report.improvementSuggestions.push_back("Add section: " + missing);
52 }
53 report.confidenceLevel = 0.75;
54 return report;
55 }

64 double AIDocumentationAnalyzer::assessCompletenessCoverage(
65 const std::vector<std::string>& docSections, const std::vector<std::string>& codeElements) {
66 if (codeElements.empty()) return 1.0;
67 std::size_t matched = 0;
68 for (const auto& element : codeElements)
69 if (std::find(docSections.begin(), docSections.end(), element) != docSections.end()) ++matched;
70 return static_cast<double>(matched) / codeElements.size();
71 }
