34template <typename Operation>
35void benchmark(const char* name, int iterations, Operation operation) {
36 // Warm up allocations and code paths before taking a measurement.
37 for (int i = 0; i < 1000; ++i) {
38 operation(i);
39 }
40 std::size_t checksum = 0;
41 const auto start = std::chrono::steady_clock::now();
42 for (int i = 0; i < iterations; ++i) {
43 checksum += operation(i);
44 }
45 const auto elapsed = std::chrono::steady_clock::now() - start;
46 const double milliseconds = std::chrono::duration<double, std::milli>(elapsed).count();
47 const double nanoseconds = std::chrono::duration<double, std::nano>(elapsed).count();
48 std::cout << std::left << std::setw(26) << name
49 << std::right << std::setw(12) << milliseconds << " ms total | "
50 << std::setw(12) << nanoseconds / iterations << " ns/op | checksum="
51 << checksum << '\n';
52}
