110 std::future<DocumentationOrchestrator::GenerationResult>
111 DocumentationOrchestrator::generateDocumentation(const DocumentationRequest& request) {
112 const auto start = std::chrono::steady_clock::now();
113 std::promise<GenerationResult> promise;
114 auto future = promise.get_future();
115 // Compute immediately and return a ready future; no background thread is needed.
116 try {
117 if (request.sourceCodePath.empty() || request.requiredFormats.empty())
118 throw std::invalid_argument("A source path and at least one format are required.");
119 if (request.qualityThreshold < 0 || request.qualityThreshold > 100)
120 throw std::invalid_argument("Quality threshold must be between 0 and 100.");
121
122 const auto provider = selectOptimalProvider(request);
123 if (provider.empty()) throw std::runtime_error("No mock provider registered.");
124 auto engine = engines.find(request.preferredProvider);
125 if (engine == engines.end()) engine = engines.begin();
126
127 static std::atomic<unsigned long long> nextId{ 0 };
128 GenerationResult result{};
129 result.requestId = "mock-" + std::to_string(++nextId);
130 result.overallQualityScore = 80.0;
131 result.warnings.push_back("Mock output: no source files or AI services were accessed.");
132 result.warnings.push_back("Formats are text placeholders, including PDF; scores are simulated.");
133 if (engine->first != request.preferredProvider)
134 result.warnings.push_back("Preferred provider unavailable; using " + provider + ".");
135 if (!request.customParameters.empty())
136 result.warnings.push_back("Custom parameters are ignored by this mock.");
137
138 const auto content = "Provider: " + provider + "\n" +
139 engine->second->generate(request.sourceCodePath);
140 for (const auto format : request.requiredFormats)
141 result.generatedContent[format] = converter->convert(content, format);
142 for (const auto audience : request.targetAudiences)
143 result.audienceScores[audience] = 80.0;
144 applyAudienceOptimizations(result);
145
146 if (result.overallQualityScore < request.qualityThreshold || !validateOutputQuality(result)) {
147 result.warnings.push_back("Simulated quality is below the required threshold.");
148 if (optimizer->blockLowQuality) {
149 result.generatedContent.clear();
150 result.warnings.push_back("Generated content was blocked by quality gating.");
151 }
152 }
153 result.generationTime = std::chrono::duration_cast<std::chrono::milliseconds>(
154 std::chrono::steady_clock::now() - start);
155 promise.set_value(std::move(result));
156 }
157 catch (...) {
158 promise.set_exception(std::current_exception());
159 }
160 return future;
161 }

236 IntelligentPromptManager::PromptOptimizationResult IntelligentPromptManager::optimizePrompt(
237 const std::string& basePrompt, const std::string& codeContext,
238 const std::vector<std::string>& qualityFeedback) {
239 PromptOptimizationResult result{};
240 result.optimizedPrompt = basePrompt;

242 result.optimizedPrompt += "\nCode context:\n" + element;
243 result.appliedOptimizations.push_back("Added code context (mock).");
244 }
245 for (const auto& feedback : qualityFeedback) {
246 result.optimizedPrompt += "\nFeedback: " + feedback;
247 result.appliedOptimizations.push_back("Appended feedback (mock).");
248 }
249 result.expectedQuality = predictPromptEffectiveness(result.optimizedPrompt);
250 result.confidenceMetrics["mockConfidence"] = result.optimizedPrompt.empty() ? 0.0 : 0.8;
251 return result;

254 void IntelligentPromptManager::updateTemplateEffectiveness(

268 std::string IntelligentPromptManager::selectBestTemplate(const std::string& context, Audience audience) {
269 // Context is intentionally ignored by this simple audience-based selection.
270 (void)context;
271 const PromptTemplate* best = nullptr;
272 for (const auto& item : templates)
273 if (item.targetAudience == audience && (!best || item.effectivenessScore > best->effectivenessScore))
274 best = &item;
275 return best ? best->basePrompt : std::string{};
276 }
